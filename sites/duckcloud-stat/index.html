💡 실제 작동 상황 및 연계 메커니즘:
외부(Cloudflare)에서 들어온 모든 사용자 요청을 GW1(100.64.0.1)과 GW2(100.64.0.2)가 50:50으로 나누어 받고, ModSecurity WAF로 SQLi/XSS 공격을 403 차단한 뒤 정상 요청만 내부 WireGuard VPN 메쉬로 전달합니다. 왼쪽 항목을 클릭하면 각 세부 기술의 실제 설정 코드와 상황 검증을 확인할 수 있습니다.
💻 대표 구현 코드 · Nginx Upstream & WAF 게이트 설정 (/etc/nginx/conf.d/duck-waf.conf):
upstream duck_active_mesh {
    server 100.64.0.31:30080 max_fails=2 fail_timeout=5s; # Primary t3
    # server 100.64.0.51:30080 backup;                    # Standby t5
    # server 100.64.0.81:30080 backup;                    # AWS t8 DR
}
server {
    listen 127.0.0.1:8088; # ModSecurity WAF Inspection Gate
    modsecurity on;
    modsecurity_rules_file /etc/nginx/modsec/main.conf;
    location / {
        proxy_pass http://127.0.0.1:8087; # 내부 메쉬 라우터로 전달
        proxy_set_header X-Duck-WAF "passed-gw1";
    }
}

실시간 점수 보정 알고리즘 파이썬 코드:
def calibrate_dr_score(raw_score, vm_ok, k8s_ok, db_ok, rtt_ms):
    adj = 0
    if not vm_ok:
        adj -= 50  # VM 다운 시 강력 감점
    elif k8s_ok and db_ok:
        adj += 5   # 웹+DB 완전 정상 시 가점
    if rtt_ms <= 30.0:
        adj += 3   # 저지연 직결 가점
    return max(0, min(100, raw_score + adj))

// JWT 무상태 세션 인증 미들웨어 (DR 전환 시에도 로그인 유지)
function authMiddleware(req, res, next) {
  const token = req.headers.authorization?.split(" ")[1];
  if (!token) return res.status(401).json({ error: "인증 토큰 누락" });
  try {
    req.user = jwt.verify(token, process.env.JWT_SECRET);
    next();
  } catch (err) {
    return res.status(403).json({ error: "유효하지 않은 토큰" });
  }
}

// 개인정보 AES-256-CBC 암호화 저장
function encryptRRN(plainRRN, secretKey) {
  const iv = crypto.randomBytes(16);
  const cipher = crypto.createCipheriv("aes-256-cbc", Buffer.from(secretKey, "hex"), iv);
  let encrypted = cipher.update(plainRRN, "utf8", "hex");
  encrypted += cipher.final("hex");
  return iv.toString("hex") + ":" + encrypted;
}
